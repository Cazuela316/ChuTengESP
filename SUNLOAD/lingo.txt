  if (sunx mod 2)=0 and (suny mod 2)=0 then
    set dir=0
  else if (sunx mod 2)=1 and (suny mod 2)=1 then
    set dir=1
  end if
  
  if dir=0 or dir=2 then
    if sunx=0 then
      changeSound "SM-27"

      changeSound "SM-28"
    else if sunx=4 then
      changeSound "SM-29"

    if suny=1 then
      changeSound "SM-30"
    else if suny=3 then
      changeSound "SM-31"
    else if suny=5 then
      changeSound "SM-32"
    end  if
  end if
end

    playSound name
  end if
  set oldSound=name
