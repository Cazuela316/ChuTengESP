on initGun
  global gunBullet
  set gunBullet=6
  set the puppet of sprite 9 to true
  set the type of sprite 9 to 1
  set the ink of sprite 9 to 36
  set the castNum of sprite 9 to the number of cast "cGun"
  set the visibility of sprite 9 to true
end

on moveNet
  global nageFlag
  global amiAnim
  global netX,netY
  global toriX,toriY
  global czone
  global tukamaeta
  global tAnim
  
  if tukamaeta then
    puppetSound "EF-15"
    pmsg 4
    if the locV of sprite 7>550 then
      getItem("ê∂Ç´ÇΩÉgÉäÉJÉâ")
      return TRUE
      --      exit
    end if
    
    set the visibility of sprite 9 to false
    set the castNum of sprite 7 to the number of cast "tukamaeta"+tAnim
    set tAnim=tAnim+1
    if tAnim>4 then set tAnim=0
    
    set the locV of sprite 7 to the locV of sprite 7 +15
    exit
  end if
  
  if the mouseDown and (not nageFlag) then
    puppetsound "19"
    set nageFlag=true
    set amiAnim=0
  end if
  
  if not nageFlag then
    set netX=the mouseH
    set netY=the mouseV
  end if
  
  if nageFlag then set amiAnim=amiAnim+1
  
  if amiAnim>7 and abs(netX-toriX)<czone and abs(netY-toriY)<czone then
    set tukamaeta=1
  else if amiAnim=10 then
    puppetSound "EF-15"
    pmsg random(3)
  end if
  
  if amiAnim>10 then
    set nageFlag=false
    set amiAnim=0
    set czone=czone+1
    if czone>80 then set czone=80
  end if
  
  set the locH of sprite 9 to netX
  set the locV of sprite 9 to netY
  set the castNum of sprite 9 to the number of cast "HIKARIAMI 0000"+amiAnim
  
  return FALSE
end

on moveGun
  global nageFlag
  global amiAnim
  global netX,netY
  global toriX,toriY
  global czone
  global tukamaeta
  global tAnim
  global gunBullet
  
  if the mouseDown and (not nageFlag) then
    set gunBullet=gunBullet-1
    set nageFlag=true
    set amiAnim=0
  end if
  
  if not nageFlag then
    set netX=the mouseH
    set netY=400
  end if
  
  if nageFlag then set amiAnim=amiAnim+1
  
  if amiAnim>7 and abs(netX-toriX)<czone and abs(netY-toriY)<czone then
    --    alert "aaa"
    set tukamaeta=1
  end if
  
  if amiAnim=10 then
    pmsg 8+random(3)
    puppetSound "EF-15"
  end if
  
  if amiAnim>10 then
    set nageFlag=false
    set amiAnim=0
    set czone=czone+1
    if czone>80 then set czone=80
  end if
  
  set the locH of sprite 9 to netX
  set the locV of sprite 9 to 350-amiAnim*10
  set the castNum of sprite 9 to the number of cast "cGun"+amiAnim
  
  if gunBullet<1 then
    return TRUE
  else
    return FALSE
  end if
  
  
end



on moveArrow
  global nageFlag
  global amiAnim
  global netX,netY
  global toriX,toriY
  global czone
  global tukamaeta
  global tAnim
  
  if the mouseDown and (not nageFlag) then
    set nageFlag=true
    set amiAnim=0
  end if
  
  if not nageFlag then
    set netX=the mouseH
    set netY=400
  end if
  
  if nageFlag then set amiAnim=amiAnim+1
  
  if amiAnim>7 and abs(netX-toriX)<czone and abs(netY-toriY)<100 then
    set tukamaeta=1
  else if amiAnim=10 then
    pmsg random(3)
    puppetSound "EF-15"
  end if
  
  if amiAnim>10 then
    set nageFlag=false
    set amiAnim=0
    set czone=czone+5
    if czone>120 then set czone=120
  end if
  
  set the locH of sprite 9 to netX
  set the locV of sprite 9 to 350-amiAnim*10
  set the castNum of sprite 9 to the number of cast "cArrow"+amiAnim
  if tukamaeta then
    return TRUE
  else
    return FALSE
  end if
end
