on goFMov label,lmovie
  global rtmovie
  global rootdir
  global ulflag
  
  if ulflag then unload
  
  cursoroff
  
  set rtmovie=the movie
  if lmovie contains ":" then
    if label="" then
      go to movie rootdir & pathtodos(lmovie)
    else
      go to label of movie rootdir & pathtodos(lmovie)
    end if
  else
    if label="" then
      go to movie lmovie
    else
      go to label of movie lmovie
    end if
  end if
end

    end if
  else
    if label="" then
      play movie lmovie
    else
      play label of movie lmovie
    end if
  end if
end

on pushPuppet
  global pptLocH,pptLocV
  global pptInk,pptType
  global pptVis,pptCastNum
  global pptppt
  
  repeat with i=1 to 48
    setAt pptLocH,i,the locH of sprite i
    setAt pptLocV,i,the locV of sprite i
    setAt pptInk,i,the ink of sprite i
    setAt pptType,i,the type of sprite i
    setAt pptVis,i,the visibility of sprite i
    setAt pptCastNum,i,the castNum of sprite i
    setAt pptppt,i,the puppet of sprite i
  end repeat
end

on popPuppet
  global pptLocH,pptLocV
  global pptInk,pptType
  global pptVis,pptCastNum
  global pptppt
  global sndBusy,sndName
  
  repeat with i=1 to 48
    set the locH of sprite i = getAt(pptLocH,i)
    set the locV of sprite i = getAt(pptLocV,i)
    set the ink of sprite i = getAt(pptInk,i)
    set the type of sprite i = getAt(pptType,i)
    set the visibility of sprite i = getAt(pptVis,i)
    set the castNum of sprite i = getAt(pptCastNum,i)
    set the puppet of sprite i = getAt(pptppt,i)
  end repeat
  
  if sndBusy=TRUE then puppetSound sndName
end
