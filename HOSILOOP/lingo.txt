on nextPoint
  global pictCast
  set the movieRate of sprite 35 to 0
  
  set xflag=true
  repeat with i=the number of cast "f0015.dif" to the number of cast "f0240.dif"
    set dummy=(char 2 to 5 of the name of cast i) *5
    if dummy>=the duration of cast "starf.mov" then set dummy=the duration of cast "starf.mov"-5
    
    if the movieTime of sprite 35<dummy then
      set the movieRate of sprite 35 to 1
      set xflag=false
      
      repeat while  (the movieTime of sprite 35)<dummy
        updateStage
      end repeat
      set the movieRate of sprite 35 to 0
      exit repeat
    end if
  end repeat
  
  if xflag then
    set the movieTime of sprite 35 to 0
    if the movieTime of sprite 35<15*5 then
      set the movieRate of sprite 35 to 1
      repeat while the movieTime of sprite 35<15*5
        updateStage
      end repeat
    end if
    set the movieRate of sprite 35 to 0
    set i=the number of cast "F0015.dif"
  end if
  
  set pictCast=the name of cast i
  revPictCast
end
